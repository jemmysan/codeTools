1- Connexion server Mysql dans Docker

    mysql -u root -p

2- Mot de passe

    root_password

3- Creer un utilisateur;

    CREATE USER 'jemmy'@'%' IDENTIFIED BY 'jemmy_password';

4- Accorder des privileges 

    GRANT ALL PRIVILEGES ON annuaire_startup.* TO 'jemmy'@'%';

    FLUSH PRIVILEGES;

5- Voir un utilisateur

    SELECT User FROM mysql.user WHERE User='nom_utilisateur';

6- Voir les identifiants d'un user 

    SELECT user, host, authentication_string FROM mysql.user WHERE user = 'jemmy';

7- Voir les bases de données

    SHOW DATABASES;
- Utiliser une base de données
    USE database_name

8- Creer une base de données 

    CREATE DATABASE database_name;

9- Voir les tables 

10- Voir les attibuts d'une table 
    DESCRIBE database_name;

11- Voir les enregistrements d'une table
    SELECT * FROM database_name

12- Supprimer une table
    DROP TABLE IF EXISTS your_table_name;
